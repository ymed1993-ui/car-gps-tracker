async function loadHistory() { try { const response = await fetch( HISTORY_URL + "?t=" + Date.now(), { cache: "no-store" } ); if (!response.ok) { return; } const history = await response.json(); if (!history) { return; } const data = Object.values(history) .filter(item => typeof item.latitude === "number" && typeof item.longitude === "number" && typeof item.timestamp === "number" && (typeof item.accuracy !== "number" || item.accuracy <= 50) ) .sort((a, b) => a.timestamp - b.timestamp); const points = []; let previousPoint = null; for (const item of data) { const currentPoint = L.latLng( item.latitude, item.longitude ); if (previousPoint) { const distance = previousPoint.distanceTo(currentPoint); /* Если новая точка внезапно дальше 300 метров от предыдущей — считаем её ошибкой GPS. */ if (distance > 300) { continue; } } points.push([ item.latitude, item.longitude ]); previousPoint = currentPoint; } routeLine.setLatLngs(points); } catch (error) { console.log( "Ошибка загрузки истории:", error ); } }
